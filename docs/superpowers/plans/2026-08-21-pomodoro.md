# Pomodoro Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Dodati `sources/pomodoro.html` — alat za rad u krugovima, s četiri zamjenjiva prikaza, popisom zadataka i tjednom mrežom.

**Architecture:** Jedna samostalna HTML datoteka. Unutra: čista jezgra bez dodira s DOM-om (`JEZGRA`), četiri prikaza iza ugovora `mount`/`paint`, dva neovisna sloja (zadaci, tjedan) i zajednički rub (zvuk, obavijesti, wake lock, jezici, teme). Build ne treba mijenjati — alat se uključuje po postojećim pravilima čim datoteka postoji.

**Tech Stack:** Vanilla HTML/CSS/JS, bez ovisnosti. Web Audio, Service Worker (ubacuje ga build), Wake Lock, `localStorage`. Node 24 za build i testove.

**Spec:** [`docs/superpowers/specs/2026-08-21-pomodoro-design.md`](../specs/2026-08-21-pomodoro-design.md)

```html
<script id="jezgra">
/* ═══════════════════════════════════════════════════════════
   JEZGRA

   Ne dira DOM i ne čita `Date.now()` sama od sebe — vrijeme joj se
   predaje. Zato se dade izvršiti u Nodeu i testirati, iako alat
   ostaje jedna samostalna datoteka.
   ═══════════════════════════════════════════════════════════ */
const JEZGRA = (() => {

  const DEFAULTS = {
    rad: 25, kratka: 5, duga: 15, ciklus: 4,
    prikaz: 'traka',
    zadaciOn: true, tjedanOn: true,
    zvuk: true, tik: true,
    obavijesti: false, budan: true,
    autoPauza: true, autoRad: false,
    tema: 'dan', lang: null
  };

  /** Plan jednog ciklusa: rad, kratka, … rad, duga. */
  function plan(cfg) {
    const faze = [];
    for (let i = 0; i < cfg.ciklus; i++) {
      faze.push({ faza: 'rad', min: cfg.rad });
      faze.push(i === cfg.ciklus - 1
        ? { faza: 'duga', min: cfg.duga }
        : { faza: 'kratka', min: cfg.kratka });
    }
    return faze;
  }

  const trajanje = (cfg, faza) =>
    ({ rad: cfg.rad, kratka: cfg.kratka, duga: cfg.duga }[faza] || 0) * 60000;

  /**
   * Lokalni datum, ne `toISOString()`.
   *
   * `toISOString` je UTC, pa bi se dnevni brojač u našoj zoni
   * resetirao u dva ujutro.
   */
  function danas(d = new Date()) {
    const pad = (n) => String(n).padStart(2, '0');
    return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`;
  }

  /** Koja faza dolazi iza dane, i s kojim brojem kruga. */
  function sljedeca(cfg, faza, krug) {
    if (faza === 'rad') {
      return krug >= cfg.ciklus - 1 ? { faza: 'duga', krug } : { faza: 'kratka', krug };
    }
    if (faza === 'kratka') return { faza: 'rad', krug: krug + 1 };
    return { faza: 'rad', krug: 0 };
  }

  /** Položaj unutar cijelog plana — treba samo prikazu „traka". */
  function udioSeanse(cfg, krug, faza, udio) {
    const faze = plan(cfg);
    const ukupno = faze.reduce((zbroj, x) => zbroj + x.min, 0);
    const index = faza === 'rad' || faza === null ? krug * 2 : krug * 2 + 1;
    let prije = 0;
    for (let i = 0; i < index; i++) prije += faze[i].min;
    return (prije + faze[index].min * (faza === null ? 0 : udio)) / ukupno;
  }

  function stanje(cfg, run, sada) {
    if (!run.faza) {
      return { mirno: true, pauzirano: false, faza: null, preostalo: 0, udio: 0,
               krug: run.krug, udioSeanse: udioSeanse(cfg, run.krug, null, 0) };
    }
    const puno = trajanje(cfg, run.faza);
    const preostalo = Math.max(0, run.ostatak != null ? run.ostatak : run.endsAt - sada);
    const udio = puno ? 1 - preostalo / puno : 1;
    return { mirno: false, pauzirano: run.ostatak != null, faza: run.faza,
             preostalo, udio, krug: run.krug,
             udioSeanse: udioSeanse(cfg, run.krug, run.faza, udio) };
  }

  return { DEFAULTS, plan, trajanje, danas, sljedeca, udioSeanse, stanje };
})();
</script>
```

```js
const STORE = 'workshop.pomodoro';

let cfg = { ...JEZGRA.DEFAULTS };
let run = { faza: null, krug: 0, endsAt: null, ostatak: null, zadatak: null };
let zadaci = [];
let dani = {};

try {
  const spremljeno = JSON.parse(localStorage.getItem(STORE) || '{}');
  cfg = { ...JEZGRA.DEFAULTS, ...(spremljeno.cfg || {}) };
  run = { ...run, ...(spremljeno.run || {}) };

  dani = spremljeno.dani && typeof spremljeno.dani === 'object' ? spremljeno.dani : {};
} catch { /* prvi put, ili neispravan zapis */ }

/** Zapisi stariji od 60 dana ne trebaju nikome. */
function orezi() {
  const granica = new Date();
  granica.setDate(granica.getDate() - 60);
  const najstariji = JEZGRA.danas(granica);
  for (const dan of Object.keys(dani)) if (dan < najstariji) delete dani[dan];
}

const save = () => {
  orezi();
  try { localStorage.setItem(STORE, JSON.stringify({ cfg, run, zadaci, dani })); }
  catch { /* privatni način */ }
```

```js
let backstop = null;

function naoruzaj() {
  clearTimeout(backstop);
  if (!run.faza || run.ostatak != null) return;
  // Rezerva za slučaj da petlja ne dođe do izraza (kartica u pozadini).
  backstop = setTimeout(zavrsiFazu, Math.max(0, run.endsAt - Date.now()) + 40);
}

function kreni(faza = run.faza || 'rad') {
  wakeAudio();
  run.faza = faza;
  run.endsAt = Date.now() + (run.ostatak != null ? run.ostatak : JEZGRA.trajanje(cfg, faza));
  run.ostatak = null;
  naoruzaj();
  zakaziZvuk();
  syncLive();
  drziBudan();
  save();
  paint();
}

function pauziraj() {
  if (!run.faza || run.ostatak != null) return;
  run.ostatak = Math.max(0, run.endsAt - Date.now());
  clearTimeout(backstop);
  otkaziZvuk();
  syncLive();
  pustiBudan();
  save();
  paint();
}

/** Preskakanje ne broji krug — nisi ga odradio. */
function preskoci() {
  if (!run.faza) return;
  const dalje = JEZGRA.sljedeca(cfg, run.faza, run.krug);
  otkaziZvuk();
  run.faza = dalje.faza;
  run.krug = dalje.krug;
  run.ostatak = null;
  kreni(dalje.faza);
}

function reset() {
  otkaziZvuk();
  clearTimeout(backstop);
  run = { faza: null, krug: 0, endsAt: null, ostatak: null, zadatak: run.zadatak };

  save();
```

- [ ] **Step 3: Kraj faze — s branom protiv dvostrukog brojanja**

```js
/**
 * Poziva se s tri strane: iz petlje, iz `backstop` timeouta i pri
 * povratku na karticu. Sva tri mogu okinuti u istoj sekundi, pa se
 * `endsAt` odmah na ulazu uzme i poništi — drugi poziv zatekne `null`
 * i tiho izađe. Bez toga jedan kraj faze zabilježi dva kruga.
 */
function zavrsiFazu() {
  const bio = run.faza;
  if (!bio || run.endsAt == null || run.ostatak != null) return;
  run.endsAt = null;
  clearTimeout(backstop);

  if (bio === 'rad') {
    const dan = JEZGRA.danas();
    dani[dan] = (dani[dan] || 0) + 1;
    const zadatak = zadaci.find((z) => z.id === run.zadatak);
    if (zadatak) zadatak.ucinjeno++;
  }

  const dalje = JEZGRA.sljedeca(cfg, bio, run.krug);
  run.faza = dalje.faza;
  run.krug = dalje.krug;
  run.ostatak = null;

  javi(bio, dalje.faza);

  const sam = dalje.faza === 'rad' ? cfg.autoRad : cfg.autoPauza;
  if (sam) kreni(dalje.faza);
  else { pustiBudan(); syncLive(); save(); paint(); }
}
```

- [ ] **Step 4: Povratak nakon izbivanja**

```js
/**
 * Vratiš se nakon tri sata: obradi se **točno jedna** istekla faza, pa
 * se stane. Namotavanje kroz cijelo razdoblje ujutro bi javilo trideset
 * odrađenih krugova kojih nema — pomodoro se broji kad ga odradiš.
 */
function uhvatiPropusteno() {
  if (!run.faza || run.ostatak != null || run.endsAt == null) return;
  if (run.endsAt > Date.now()) return;

  const zaostalo = Date.now() - run.endsAt;
  zavrsiFazu();
  // Faza koja je krenula sama također je već istekla — stani.
  if (run.endsAt != null && run.endsAt <= Date.now()) {
    run.ostatak = null;
    run.endsAt = null;
    clearTimeout(backstop);
    pustiBudan();
    save();
  }
  if (zaostalo > 60_000) poruka(t('istekloDokTeNijeBilo'));
  paint();
}

addEventListener('visibilitychange', () => { if (!document.hidden) uhvatiPropusteno(); });
```

```js
let audio = null;
let zakazani = [];

function wakeAudio() {
  if (!audio) audio = new (window.AudioContext || window.webkitAudioContext)();
  if (audio.state === 'suspended') audio.resume();
}

/** Jedan ton; `delay` je odmak u sekundama od sada. */
function ton(freq, sekundi, glasnoca = 0.22, delay = 0) {
  if (!cfg.zvuk || !audio) return;
  const t0 = audio.currentTime + delay;
  const osc = audio.createOscillator();
  const gain = audio.createGain();
  osc.frequency.value = freq;
  gain.gain.setValueAtTime(0, t0);
  gain.gain.linearRampToValueAtTime(glasnoca, t0 + 0.01);
  gain.gain.exponentialRampToValueAtTime(0.0001, t0 + sekundi);
  osc.connect(gain).connect(audio.destination);
  osc.start(t0);
  osc.stop(t0 + sekundi + 0.05);
  zakazani.push(osc);
}
```

- [ ] **Step 2: Zakazivanje unaprijed**

```js
/**
 * Signal kraja se zakazuje čim faza počne.
 *
 * To je jedini način da zvuk stigne dok je kartica prigušena u
 * pozadini — `setTimeout` ondje kasni, ali `AudioContext` ima svoj
 * sat. Isto radi `timer.html` u `scheduleSounds()`.
 */
function zakaziZvuk() {
  otkaziZvuk();
  if (!cfg.zvuk || !audio || !run.faza || run.ostatak != null) return;
  const doKraja = (run.endsAt - Date.now()) / 1000;
  if (doKraja <= 0) return;

  if (cfg.tik) for (let i = 3; i >= 1; i--) {
    if (doKraja - i > 0) ton(880, 0.06, 0.12, doKraja - i);
  }
  if (run.faza === 'rad') {           // kraj rada — topliji, dvotonski
    ton(523.25, 0.45, 0.26, doKraja);
    ton(659.25, 0.6, 0.22, doKraja + 0.16);
  } else {                            // kraj pauze — kraći, jedan ton
    ton(784, 0.35, 0.24, doKraja);
  }
}

    za fazu koje više nema. */
```

- [ ] **Step 2: Trajna obavijest sa satom**

```js
const OZNAKA = 'workshop-pomodoro';
let potpis = null;

/**
 * Obavijest nosi **sat u koliko faza završava**, ne odbrojavanje.
 * Izračuna se jednom i ne stari, pa se šalje samo kad se stanje
 * stvarno promijeni. Osvježavanje svake sekunde Chrome prijavljuje kao
 * zlouporabu obavijesti.
 */
async function syncLive() {
  if (!cfg.obavijesti || !navigator.serviceWorker) return;
  const reg = await navigator.serviceWorker.ready;

  if (!run.faza || run.ostatak != null) {
    const noviPotpis = `${run.faza || 'mirno'}|pauza`;
    if (noviPotpis === potpis) return;
    potpis = noviPotpis;
  } else {
    const kraj = new Date(run.endsAt).toLocaleTimeString(LOCALES[lang] || 'hr-HR',
      { hour: '2-digit', minute: '2-digit' });
    const zadatak = zadaci.find((z) => z.id === run.zadatak);
    const naslov = `${t(run.faza)} ${t('do')} ${kraj}`;
    const tijelo = zadatak ? zadatak.tekst : t('bezZadatka');
    const noviPotpis = `${naslov}|${tijelo}`;
    if (noviPotpis === potpis) return;      // ista poruka se ne šalje dvaput
    potpis = noviPotpis;

    // Tipke postoje samo ondje gdje ih platforma podržava.
    const akcije = (Notification.maxActions || 0) >= 2
      ? [{ action: 'pauza', title: t('pauza') },
         { action: 'skip', title: t('preskoci') },
         { action: 'stop', title: t('zaustavi') }].slice(0, Notification.maxActions)
      : [];

    return reg.showNotification(naslov, {
      body: tijelo, tag: OZNAKA, renotify: false, silent: true,
      requireInteraction: true, actions: akcije,
      data: { url: location.pathname }
    });
  }
  ugasiObavijest();
}

async function ugasiObavijest() {
  if (!navigator.serviceWorker) return;
  const reg = await navigator.serviceWorker.ready;
  for (const n of await reg.getNotifications({ tag: OZNAKA })) n.close();
}
```
